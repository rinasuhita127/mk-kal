import numpy as np
A = np.array([[3, 1, 0],[1, 3, 0],[0, 0, 2]])
U, S, VT = np.linalg.svd(A)
Sigma = np.diag(S)
hasil = U @ Sigma @ VT
print("A =")
print(A)
print("U =")
print(U)
print("S =")
print(S)
print("Sigma =")
print(Sigma)
print("VT =")
print(VT)
print(hasil)
