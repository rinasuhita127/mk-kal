import numpy as np
A = np.array([[1, 2, 0],[0, 1, 1],[2, 0, 1]])
U, S, VT = np.linalg.svd(A)
Sigma = np.diag(S)
hasil = U @ Sigma @ VT
print("A =")
print(A)
print("U =")
print(U)
print("S =")
print(S)
print("Sigma =")
print(Sigma)
print("VT =")
print(VT)
print("Hasil =")
print(hasil)
